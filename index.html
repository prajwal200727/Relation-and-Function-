def is_reflexive():
    return all((a, a) in R for a in A)

def is_irreflexive():
    return all((a, a) not in R for a in A)

def is_symmetric():
    return all((b, a) in R for (a, b) in R)

def is_antisymmetric():
    return all(a == b for (a, b) in R if (b, a) in R)

def is_asymmetric():
    return all((b, a) not in R for (a, b) in R)

def is_transitive():
    for (a, b) in R:
        for (c, d) in R:
            if b == c and (a, d) not in R:
                return False
    return True

def is_function():
    # Each element in A must appear as first element exactly once
    from collections import Counter
    counts = Counter(a for (a, b) in R)
    return all(counts[a] == 1 for a in A) and len(R) == len(A)

def is_injective():
    if not is_function(): return False
    images = [b for (a, b) in R]
    return len(images) == len(set(images))

def is_surjective():
    if not is_function(): return False
    return set(A).issubset({b for (a, b) in R})

def is_bijective():
    return is_injective() and is_surjective()
